const renderedChatMessages=new WeakMap();
function escapeChatText(value){
  return String(value??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}
function chatMessagesHtml(messages){
  return [...(messages||[])].reverse().map(m=>{
    const t=new Date(m.timestamp).toLocaleString('es-CO',{hour:'2-digit',minute:'2-digit',day:'2-digit',month:'short'});
    const isIn=m.direccion==='entrada';
    const intentTag=m.intent?'<div class="msg-intent">'+escapeChatText(m.intent)+'</div>':'';
    return '<div class="msg '+(isIn?'msg-in':'msg-out')+'"><div>'+escapeChatText(m.contenido||'')+'</div><div class="msg-meta">'+(isIn?'Cliente':'Sofia')+' · '+t+'</div>'+intentTag+'</div>';
  }).join('');
}
function rememberChatMessages(container,html){
  if(container)renderedChatMessages.set(container,html);
}
function refreshOpenChat(){
  const container=document.getElementById('chat-msgs-inner');
  if(!currentTel||!container||!Array.isArray(grouped[currentTel]))return;
  const html=chatMessagesHtml(grouped[currentTel]);
  if(renderedChatMessages.get(container)===html)return;
  const oldTop=container.scrollTop;
  const followLatest=container.scrollHeight-container.clientHeight-oldTop<=80;
  container.innerHTML=html;
  rememberChatMessages(container,html);
  container.scrollTop=followLatest?container.scrollHeight:oldTop;
}
